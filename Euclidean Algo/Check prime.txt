for prime check opposite to number of divisors question


int prime(int n){
    int i;
    for (i = 2;i<n;i++){
        if (n%i==0){
            return false
        }
    }
    return true;
}



optimized version


int prime(int n){
    int i;
    for (i = 2;(i*i)<n;i++){
        if ((n/i)%i==0){
            return false
        }
    }
    return true;
}
